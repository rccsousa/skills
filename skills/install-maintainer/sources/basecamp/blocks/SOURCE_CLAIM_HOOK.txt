          if [ -n "${BASECAMP_TOKEN:-}" ] && command -v basecamp >/dev/null 2>&1; then
            body="$(gh issue view "$ISSUE" --json body --jq '.body' 2>/dev/null || true)"
            card_id="$(printf '%s' "$body" | grep -oE 'basecamp-card:[0-9]+' | head -n1 | cut -d: -f2 || true)"
            if [ -n "$card_id" ]; then
              echo "Moving Basecamp card ${card_id} to 'in progress'."
              basecamp cards move "$card_id" \
                --to "$BASECAMP_PROGRESS_COLUMN" \
                --card-table "$BASECAMP_CARD_TABLE" \
                --in "$BASECAMP_PROJECT" || echo "card move failed (non-fatal)."
            else
              echo "No basecamp-card marker in issue #${ISSUE} — skipping board move."
            fi
          fi
